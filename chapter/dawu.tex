\section{质点运动学}


\begin{enumerate}

\item 质点沿半径为$R$的圆周做匀速率运动,每间隔时间$T$转一圈.在$2T$时间间隔中,其平均速度大小和平均速率分别为(\quad)

\begin{choices}
	\item $2\pi R/T,2\pi R/T$
	\item $0,2\pi R/T$
	\item $0,0$
	\item $2\pi R/T,0$
\end{choices}


\item 质点做半径为$R$的变速圆周运动时的加速度大小为\footnote{$v$表示任一时刻质点的速率}(\quad)

\begin{choices}
	\item $\frac{dv}{dt}$
	\item $\frac{v^2}{R}$
	\item $\frac{dv}{dt} + \frac{v^2}{R}$
	\item $\sqrt[2]{(\frac{dv}{dt}) ^ 2 + (\frac{v^2}{R}) ^ 2}$
\end{choices}


\item 质点沿半径为$R$的圆周运动,运动学方程为$\theta  = 3 + 2t^2$,则$t$时刻质点的法向加速度的大小为$a_n = \underline{\hspace{6em}}.$

\item 已知质点的位置矢量随时间变化的函数形式为$\vec{r} = 4t^{2} \vec{i} + (3 + 2t) \vec{j}$\footnote{SI单位}


\end{enumerate}
